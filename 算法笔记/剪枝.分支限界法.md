# 分支限界法

1. 用分支限界法求解旅行商问题。

```python
import heapq
import copy

class Node:
    """解空间树的节点"""
    def __init__(self, matrix, cost, bound, path, curr_city):
        self.matrix = matrix      # 当前节点的归约矩阵
        self.cost = cost          # 当前已确定的边权重之和
        self.bound = bound        # 当前节点的下界（估算最小值）
        self.path = path          # 当前已走过的路径列表
        self.curr_city = curr_city  # 当前所在的城市
        
    def __lt__(self, other):
        # 优先队列（小顶堆）依据 bound 从小到大排序
        return self.bound < other.bound

def reduce_matrix(matrix):
    """
    对矩阵进行归约操作，并返回归约后的矩阵和本轮减少的成本值（归约常数）
    """
    n = len(matrix)
    reduced_matrix = copy.deepcopy(matrix)
    reduction_cost = 0

    # 1. 行归约
    for i in range(n):
        min_val = min(reduced_matrix[i])
        if min_val != float('inf') and min_val > 0:
            reduction_cost += min_val
            for j in range(n):
                if reduced_matrix[i][j] != float('inf'):
                    reduced_matrix[i][j] -= min_val

    # 2. 列归约
    for j in range(n):
        min_val = min(reduced_matrix[i][j] for i in range(n))
        if min_val != float('inf') and min_val > 0:
            reduction_cost += min_val
            for i in range(n):
                if reduced_matrix[i][j] != float('inf'):
                    reduced_matrix[i][j] -= min_val

    return reduced_matrix, reduction_cost

def solve_tsp(graph):
    n = len(graph)
    
    # 初始化根节点
    init_matrix, init_reduction = reduce_matrix(graph)
    
    root = Node(
        matrix=init_matrix,
        cost=0,
        bound=init_reduction,
        path=[0],
        curr_city=0
    )
    
    # 创建优先队列并将根节点入队
    pq = []
    heapq.heappush(pq, root)
    
    best_cost = float('inf')
    best_path = []
    
    while pq:
        # 取出当前下界（bound）最小的节点
        curr_node = heapq.heappop(pq)
        
        # 剪枝：如果当前节点的下界已经大于等于已知最优解，后续分支无需再看
        if curr_node.bound >= best_cost:
            continue
            
        # 如果路径包含了所有城市，说明找到了一个完整解
        if len(curr_node.path) == n:
            # 加上回到起点的距离（如果存在的话）
            last_city = curr_node.curr_city
            return_cost = graph[last_city][0]
            
            if return_cost != float('inf'):
                total_cost = curr_node.cost + return_cost
                if total_cost < best_cost:
                    best_cost = total_cost
                    best_path = curr_node.path + [0]
            continue
            
        # 扩展当前节点：尝试去往下一个未访问的城市
        curr_city = curr_node.curr_city
        for next_city in range(n):
            # 不能去已经去过的城市，且图上必须有通路
            if next_city not in curr_node.path and curr_node.matrix[curr_city][next_city] != float('inf'):
                
                # 复制并准备计算子节点的矩阵
                child_matrix = copy.deepcopy(curr_node.matrix)
                
                # 记录从当前城市到下一个城市的边权重
                edge_cost = curr_node.matrix[curr_city][next_city]
                
                # 矩阵变换规则：
                # 1. 离开的城市行设为 inf
                for j in range(n): child_matrix[curr_city][j] = float('inf')
                # 2. 进入的城市列设为 inf
                for i in range(n): child_matrix[i][next_city] = float('inf')
                # 3. 封锁反向边，防止提前成环（禁止直接回到起点）
                child_matrix[next_city][0] = float('inf')
                
                # 对子矩阵进行归约，获取新的归约常数
                reduced_child_matrix, child_reduction = reduce_matrix(child_matrix)
                
                # 计算子节点的下界：父节点下界 + 走这条边的实际消耗 + 新矩阵的归约消耗
                child_bound = curr_node.bound + edge_cost + child_reduction
                child_cost = curr_node.cost + graph[curr_city][next_city]
                
                # 剪枝判断
                if child_bound < best_cost:
                    child_node = Node(
                        matrix=reduced_child_matrix,
                        cost=child_cost,
                        bound=child_bound,
                        path=curr_node.path + [next_city],
                        curr_city=next_city
                    )
                    heapq.heappush(pq, child_node)
                    
    return best_path, best_cost

```

2. 用分支限界法求解员工分配问题。

```python
def min_cost_assignment(cost):

    import heapq
    n = len(cost)
    # 定义节点类
    class Node:
        def __init__(self, level, assignment, cost):
            self.level = level          # 当前分配到第几个人了
            self.assignment = assignment  # 当前的分配方案
            self.cost = cost            # 当前的总代价

        def __lt__(self, other):
            return self.cost < other.cost  # 优先队列根据总代价排序
    # 计算当前分配方案的总代价
    def calculate_cost(assignment):
        total_cost = 0
        for i in range(len(assignment)):
            if assignment[i] != -1:
                total_cost += cost[i][assignment[i]]
        return total_cost
    # 初始化优先队列
    pq = []
    initial_node = Node(level=0, assignment=[-1] * n, cost=0)
    heapq.heappush(pq, initial_node)
    best_cost = float('inf')
    best_assignment = None

    while pq:
        current_node = heapq.heappop(pq)
        if current_node.level == n:  # 已经分配完所有人了
            if current_node.cost < best_cost:
                best_cost = current_node.cost
                best_assignment = current_node.assignment
            continue
        # 尝试给当前人分配每个任务
        for task in range(n):
            if task not in current_node.assignment:  # 任务还没有被分配
                new_assignment = current_node.assignment.copy()
                new_assignment[current_node.level] = task
                new_cost = calculate_cost(new_assignment)
                if new_cost < best_cost:  # 剪枝：如果当前总代价已经超过最优解了，就不继续扩展了
                    new_node = Node(level=current_node.level + 1, assignment=new_assignment, cost=new_cost)
                    heapq.heappush(pq, new_node)

    return best_cost, best_assignment

# 测试数据
cost = [
    [3, 5, 2, 4],
    [6, 7, 5, 3],
    [3, 7, 4, 5],
    [8, 5, 4, 6]
]
min_cost, assignment = min_cost_assignment(cost)
print("最小总代价:", min_cost)
print("分配方案:", assignment)
```

4. 用分支限界法求解n皇后问题。

答: N皇后问题没有很好的上下限剪枝方法, 所以使用普通队列进行广度优先搜索。

```python
def n_queens(n):
    queens = [-1] * n  # queens[i] = c 表示第 i 行皇后放在第 c 列
    solutions = []

    def is_safe(row, col):
        for prev_row in range(row):
            if queens[prev_row] == col or \
               queens[prev_row] - prev_row == col - row or \
               queens[prev_row] + prev_row == col + row:
                return False
        return True

    from collections import deque
    q = deque()

    # 从第0行开始
    q.append((0, queens.copy()))

    while q:
        row, current_queens = q.popleft()       # 取出队头节点
        if row == n:
            solutions.append(current_queens)    # 是叶子节点,找到一个解
            continue
        for col in range(n):                    # 添加子节点
            if is_safe(row, col):
                new_queens = current_queens.copy()
                new_queens[row] = col
                q.append((row + 1, new_queens))

    return solutions
```
