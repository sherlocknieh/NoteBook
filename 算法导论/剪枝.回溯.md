7. 设计3-CNF-SAT问题的回溯算法。

```python
def DPLL(Formula, Assignment):
    # Formula: 以子句列表形式表示的布尔公式，每个子句是一个文字列表
    # Assignment: 当前变量赋值的字典，{变量: True/False}

    # 核心步骤 1: 简化公式
    Formula = simplify(Formula, Assignment)
    
    # 基准情况：所有子句都已被满足（公式变空）
    if Formula is empty:
        return True, Assignment
        
    # 基准情况：出现了空子句（即某个子句里所有文字都为 False，无法被满足）
    if Formula contains an empty clause:
        return False, None
        
    # 核心步骤 2: 单子句传播（剪枝）
    while Formula contains a unit clause (L):
        # L 是一个孤立的文字，比如 x1 或 not x2
        Assignment = Assignment + {L: True} # 强制让它为真
        Formula = simplify(Formula, Assignment)
        if Formula is empty: return True, Assignment
        if Formula contains an empty clause: return False, None

    # 核心步骤 3: 选择一个尚未赋值的变量 x
    x = choose_unassigned_variable(Formula)
    
    # 尝试分支 1: 设 x 为 True
    success, result = DPLL(Formula, Assignment + {x: True})
    if success:
        return True, result
        
    # 回溯并尝试分支 2: 设 x 为 False
    success, result = DPLL(Formula, Assignment + {x: False})
    if success:
        return True, result
        
    # 两个分支都失败了，向上层回溯
    return False, None
```

8. 设计一个回溯算法求解汉密尔顿回路问题。

```python
def find_hamiltonian_cycle(G):
    # G: 图的邻接表表示，G[v] 是与顶点 v 相邻的顶点列表
    n = len(G)
    path = [None] * n  # 存储当前路径
    visited = [False] * n  # 记录哪些顶点已访问

    def backtrack(pos):
        # pos: 当前路径中的位置，0 <= pos < n
        if pos == n:
            # 已经访问了所有顶点，检查是否形成回路
            return path[pos - 1] in G[path[0]]
        
        for neighbor in G[pos]:
            if not visited[neighbor]:
                visited[neighbor] = True
                path[pos] = neighbor
                if backtrack(pos + 1):
                    return True
                visited[neighbor] = False  # 回溯
        return False

    # 从第一个顶点开始尝试构建路径
    visited[0] = True
    path[0] = 0
    if backtrack(1):
        return path + [path[0]]  # 返回完整的回路路径
    else:
        return None  # 没有找到汉密尔顿回路
```

10. n个雇员被分配做n件工作，cost[i][j] 表示第i个人做第j件工作的代价，设计一个回溯算法找出一种分配使得总代价最少。

```python
def min_cost_assignment(cost):
    # cost: n x n 的代价矩阵
    # 行 i 表示第 i 个人
    # 列 j 表示第 j 件工作
    # cost[i][j] 是第 i 个人做第 j 件工作的代价

    # 全局变量
    G = {
        'assigned': [], # 已经分配了哪些工作，存储工作索引
        'min_cost': float('inf'),  # 当前找到的最小总代价
    }

    def backtrack(i, current_cost, N):
        # i: 当前正在分配第 i 个人
        # current_cost: 当前已经累积的总代价
        # N: 工作总数

        # 已离开底层，更新最小总代价
        if i == N: 
            G['min_cost'] = min(G['min_cost'], current_cost)
            return

        # 遍历第 i 行的每个工作 j
        for j in range(N):
            # 如果工作 j 已经被分配了, 跳过
            if j in G['assigned']: continue
            
            # 如果当前总代价已经超过已知最小总代价，剪枝
            if current_cost + cost[i][j] >= G['min_cost']:  continue

            # 选择工作 j 给第 i 个人
            G['assigned'].append(j)
            # 递归分配下一个人
            backtrack(i + 1, current_cost + cost[i][j], N)
            # 回溯，撤销选择
            G['assigned'].pop()

    backtrack(0, 0, len(cost))  # 从第 0 个人开始分配，初始总代价为 0
    return G['min_cost']
        

# 测试
cost = [[4, 2, 3],
        [2, 3, 1],
        [3, 1, 2]]
print(f"最小总代价: {min_cost_assignment(cost)}")
# 最小代价 = 6
```
